```markdown
# Expr Language Specification

## Overview
Expr is a small dynamically-typed scripting language with variables, functions, 
conditionals, operators, lists, and error handling.

## Design Decisions
- Errors propagate as values (no exceptions)
- First-class functions with closures
- Immutable lists and records
- Semicolon-terminated statements

## Grammar (precedence: low→high)
```
program    → stmt* EOF
stmt       → let_stmt | expr_stmt
let_stmt   → "let" NAME "=" expr ";"
expr_stmt  → expr ";"

expr       → or_expr
or_expr    → and_expr ("||" and_expr)*
and_expr   → eq_expr ("&&" eq_expr)*
eq_expr    → cmp_expr (("==" | "!=") cmp_expr)*
cmp_expr   → add_expr (("<" | ">" | "<=" | ">=") add_expr)*
add_expr   → mul_expr (("+" | "-") mul_expr)*
mul_expr   → unary_expr (("*" | "/" | "%") unary_expr)*
unary_expr → ("!" | "-") unary_expr | call_expr
call_expr  → primary_expr ("(" args ")")?
primary_expr → NUMBER | STRING | NAME | "true" | "false" | "null"
             | "(" expr ")"
             | "[" (expr ("," expr)*)? "]"
             | "{" (NAME ":" expr ("," NAME ":" expr)*)? "}"
             | "if" expr "{" expr "}" "else" "{" expr "}"
             | "fn" "(" params ")" "{" expr "}"

args       → (expr ("," expr)*)?
params     → (NAME ("," NAME)*)?
```

## Exit Codes
- 0: success
- 1: program error (runtime)
- 2: parse/lex error

## Error Values
Errors are `{error: "reason"}` records. Use `missed(x)` to check if x is an error.
```

```python
import re
from dataclasses import dataclass
from typing import List, Optional

@dataclass
class Token:
    type: str
    value: any
    line: int
    col: int

class LexError(Exception):
    def __init__(self, message: str, line: int, col: int):
        self.message = message
        self.line = line
        self.col = col
    def __str__(self):
        return f"error: {self.message} at {self.line}:{self.col}"

def tokenize(src: str) -> List[Token]:
    tokens = []
    line, col = 1, 1
    i = 0
    
    keywords = {"let", "true", "false", "null", "if", "else", "fn"}
    
    while i < len(src):
        start_col = col
        
        if src[i].isspace():
            if src[i] == '\n':
                line += 1
                col = 1
            else:
                col += 1
            i += 1
            continue
        
        if src[i:i+2] == '//':
            while i < len(src) and src[i] != '\n':
                i += 1
            continue
        
        if src[i].isdigit():
            j = i
            while j < len(src) and src[j].isdigit():
                j += 1
            if j < len(src) and src[j] == '.':
                j += 1
                while j < len(src) and src[j].isdigit():
                    j += 1
            val = src[i:j]
            tokens.append(Token('NUMBER', float(val) if '.' in val else int(val), line, start_col))
            col += j - i
            i = j
            continue
        
        if src[i] == '"':
            i += 1
            j = i
            val = ""
            while j < len(src) and src[j] != '"':
                if src[j] == '\\' and j + 1 < len(src):
                    if src[j+1] == 'n':
                        val += '\n'
                    elif src[j+1] == 't':
                        val += '\t'
                    elif src[j+1] == '\\':
                        val += '\\'
                    elif src[j+1] == '"':
                        val += '"'
                    else:
                        val += src[j+1]
                    j += 2
                else:
                    val += src[j]
                    j += 1
            if j >= len(src):
                raise LexError("unterminated string", line, start_col)
            tokens.append(Token('STRING', val, line, start_col))
            col += j - i + 2
            i = j + 1
            continue
        
        if src[i].isalpha() or src[i] == '_':
            j = i
            while j < len(src) and (src[j].isalnum() or src[j] == '_'):
                j += 1
            word = src[i:j]
            if word in keywords:
                tokens.append(Token(word.upper(), word, line, start_col))
            else:
                tokens.append(Token('NAME', word, line, start_col))
            col += j - i
            i = j
            continue
        
        two_char_ops = ["==", "!=", "<=", ">=", "||", "&&"]
        if i + 1 < len(src) and src[i:i+2] in two_char_ops:
            tokens.append(Token(src[i:i+2], src[i:i+2], line, start_col))
            col += 2
            i += 2
            continue
        
        single_char_ops = "+-*/%()[]{}:;,.<>!&|"
        if src[i] in single_char_ops:
            tokens.append(Token(src[i], src[i], line, start_col))
            col += 1
            i += 1
            continue
        
        raise LexError(f'unexpected character: {src[i]}', line, start_col)
    
    tokens.append(Token('EOF', None, line, col))
    return tokens
```

## expr/parser.py

```python
from dataclasses import dataclass
from typing import List, Optional, Any
from expr.lexer import Token, tokenize, LexError

@dataclass
class ASTNode:
    line: int
    col: int

def _simple(name, fields):
    def __init__(self, line, col, **kw):
        self.line = line
        self.col = col
        for field in fields.split():
            setattr(self, field, kw.get(field))
    
    cls = type(name, (ASTNode,), {'__init__': __init__})
    return cls

Number = _simple('Number', 'value')
String = _simple('String', 'value')
Bool = _simple('Bool', 'value')
Null = _simple('Null', '')
Name = _simple('Name', 'name')
List = _simple('List', 'items')
Record = _simple('Record', 'fields')
BinOp = _simple('BinOp', 'left op right')
UnaryOp = _simple('UnaryOp', 'op operand')
Call = _simple('Call', 'func args')
If = _simple('If', 'cond then_expr else_expr')
Fn = _simple('Fn', 'params body')
Let = _simple('Let', 'name value')
Program = _simple('Program', 'stmts')

class ParseError(Exception):
    def __init__(self, message: str, line: int, col: int):
        self.message = message
        self.line = line
        self.col = col
    def __str__(self):
        return f"error: {self.message} at {self.line}:{self.col}"

class Parser:
    def __init__(self, tokens: List[Token]):
        self.tokens = tokens
        self.pos = 0
    
    def current(self) -> Token:
        return self.tokens[self.pos]
    
    def peek(self, offset=1) -> Token:
        pos = self.pos + offset
        if pos < len(self.tokens):
            return self.tokens[pos]
        return self.tokens[-1]
    
    def advance(self):
        self.pos += 1
    
    def expect(self, expected: str) -> Token:
        token = self.current()
        if token.type != expected and token.value != expected:
            raise ParseError(f"expected {expected}, got {token.type}", token.line, token.col)
        self.advance()
        return token
    
    def match(self, *types) -> bool:
        return self.current().type in types or self.current().value in types
    
    def parse(self) -> Program:
        stmts = []
        while not self.match('EOF'):
            stmts.append(self.parse_stmt())
        return Program(line=1, col=1, stmts=stmts)
    
    def parse_stmt(self):
        if self.match('LET'):
            return self.parse_let()
        return self.parse_expr_stmt()
    
    def parse_let(self):
        let_tok = self.expect('LET')
        name_tok = self.expect('NAME')
        self.expect('=')
        value = self.parse_expr()
        self.expect(';')
        return Let(line=let_tok.line, col=let_tok.col, name=name_tok.value, value=value)
    
    def parse_expr_stmt(self):
        expr = self.parse_expr()
        self.expect(';')
        return expr
    
    def parse_expr(self):
        return self.parse_or()
    
    def parse_or(self):
        left = self.parse_and()
        while self.match('||'):
            op_tok = self.current()
            self.advance()
            right = self.parse_and()
            left = BinOp(line=op_tok.line, col=op_tok.col, left=left, op='||', right=right)
        return left
    
    def parse_and(self):
        left = self.parse_eq()
        while self.match('&&'):
            op_tok = self.current()
            self.advance()
            right = self.parse_eq()
            left = BinOp(line=op_tok.line, col=op_tok.col, left=left, op='&&', right=right)
        return left
    
    def parse_eq(self):
        left = self.parse_cmp()
        while self.match('==', '!='):
            op_tok = self.current()
            op = op_tok.value
            self.advance()
            right = self.parse_cmp()
            left = BinOp(line=op_tok.line, col=op_tok.col, left=left, op=op, right=right)
        return left
    
    def parse_cmp(self):
        left = self.parse_add()
        while self.match('<', '>', '<=', '>='):
            op_tok = self.current()
            op = op_tok.value
            self.advance()
            right = self.parse_add()
            left = BinOp(line=op_tok.line, col=op_tok.col, left=left, op=op, right=right)
        return left
    
    def parse_add(self):
        left = self.parse_mul()
        while self.match('+', '-'):
            op_tok = self.current()
            op = op_tok.value
            self.advance()
            right = self.parse_mul()
            left = BinOp(line=op_tok.line, col=op_tok.col, left=left, op=op, right=right)
        return left
    
    def parse_mul(self):
        left = self.parse_unary()
        while self.match('*', '/', '%'):
            op_tok = self.current()
            op = op_tok.value
            self.advance()
            right = self.parse_unary()
            left = BinOp(line=op_tok.line, col=op_tok.col, left=left, op=op, right=right)
        return left
    
    def parse_unary(self):
        if self.match('!', '-'):
            op_tok = self.current()
            op = op_tok.value
            self.advance()
            operand = self.parse_unary()
            return UnaryOp(line=op_tok.line, col=op_tok.col, op=op, operand=operand)
        return self.parse_call()
    
    def parse_call(self):
        expr = self.parse_primary()
        while self.match('('):
            self.advance()
            args = []
            if not self.match(')'):
                args.append(self.parse_expr())
                while self.match(','):
                    self.advance()
                    args.append(self.parse_expr())
            self.expect(')')
            expr = Call(line=expr.line, col=expr.col, func=expr, args=args)
        return expr
    
    def parse_primary(self):
        tok = self.current()
        
        if tok.type == 'NUMBER':
            self.advance()
            return Number(line=tok.line, col=tok.col, value=tok.value)
        
        if tok.type == 'STRING':
            self.advance()
            return String(line=tok.line, col=tok.col, value=tok.value)
        
        if tok.type == 'TRUE':
            self.advance()
            return Bool(line=tok.line, col=tok.col, value=True)
        
        if tok.type == 'FALSE':
            self.advance()
            return Bool(line=tok.line, col=tok.col, value=False)
        
        if tok.type == 'NULL':
            self.advance()
            return Null(line=tok.line, col=tok.col)
        
        if tok.type == 'NAME':
            self.advance()
            return Name(line=tok.line, col=tok.col, name=tok.value)
        
        if tok.type == '[':
            self.advance()
            items = []
            if not self.match(']'):
                items.append(self.parse_expr())
                while self.match(','):
                    self.advance()
                    items.append(self.parse_expr())
            self.expect(']')
            return List(line=tok.line, col=tok.col, items=items)
        
        if tok.type == '{':
            self.advance()
            fields = {}
            if not self.match('}'):
                key_tok = self.expect('NAME')
                self.expect(':')
                fields[key_tok.value] = self.parse_expr()
                while self.match(','):
                    self.advance()
                    key_tok = self.expect('NAME')
                    self.expect(':')
                    fields[key_tok.value] = self.parse_expr()
            self.expect('}')
            return Record(line=tok.line, col=tok.col, fields=fields)
        
        if tok.type == 'IF':
            self.advance()
            cond = self.parse_expr()
            self.expect('{')
            then_expr = self.parse_expr()
            self.expect('}')
            self.expect('ELSE')
            self.expect('{')
            else_expr = self.parse_expr()
            self.expect('}')
            return If(line=tok.line, col=tok.col, cond=cond, then_expr=then_expr, else_expr=else_expr)
        
        if tok.type == 'FN':
            self.advance()
            self.expect('(')
            params = []
            if not self.match(')'):
                params.append(self.expect('NAME').value)
                while self.match(','):
                    self.advance()
                    params.append(self.expect('NAME').value)
            self.expect(')')
            self.expect('{')
            body = self.parse_expr()
            self.expect('}')
            return Fn(line=tok.line, col=tok.col, params=params, body=body)
        
        if tok.type == '(':
            self.advance()
            expr = self.parse_expr()
            self.expect(')')
            return expr
        
        raise ParseError(f"unexpected {tok.type}", tok.line, tok.col)

def parse(src: str):
    tokens = tokenize(src)
    parser = Parser(tokens)
    return parser.parse()
```

## expr/evaluator.py

```python
from expr.parser import *
from typing import Any, Dict, Optional

def error(reason: str) -> Dict:
    return {"error": reason}

def is_error(v: Any) -> bool:
    return isinstance(v, dict) and "error" in v and len(v) == 1

class Env:
    def __init__(self, parent=None):
        self.parent = parent
        self.vars = {}
    
    def define(self, name: str, value: Any):
        self.vars[name] = value
    
    def get(self, name: str) -> Any:
        if name in self.vars:
            return self.vars[name]
        if self.parent:
            return self.parent.get(name)
        return error(f'unbound name: {name}')
    
    def extend(self):
        return Env(parent=self)

class Closure:
    def __init__(self, params: list, body: ASTNode, env: Env):
        self.params = params
        self.body = body
        self.env = env

def eval_node(node: ASTNode, env: Env) -> Any:
    if isinstance(node, Number):
        return node.value
    if isinstance(node, String):
        return node.value
    if isinstance(node, Bool):
        return node.value
    if isinstance(node, Null):
        return None
    if isinstance(node, Name):
        return env.get(node.name)
    if isinstance(node, List):
        items = []
        for item in node.items:
            v = eval_node(item, env)
            if is_error(v):
                return v
            items.append(v)
        return items
    if isinstance(node, Record):
        record = {}
        for key, val_node in node.fields.items():
            v = eval_node(val_node, env)
            if is_error(v):
                return v
            record[key] = v
        return record
    if isinstance(node, BinOp):
        left = eval_node(node.left, env)
        if is_error(left):
            return left
        right = eval_node(node.right, env)
        if is_error(right):
            return right
        
        if node.op == '+':
            if isinstance(left, (int, float)) and isinstance(right, (int, float)):
                return left + right
            if isinstance(left, str) and isinstance(right, str):
                return left + right
            if isinstance(left, list) and isinstance(right, list):
                return left + right
            return error(f'cannot add {type(left).__name__} and {type(right).__name__}')
        if node.op == '-':
            if isinstance(left, (int, float)) and isinstance(right, (int, float)):
                return left - right
            return error(f'cannot subtract {type(right).__name__} from {type(left).__name__}')
        if node.op == '*':
            if isinstance(left, (int, float)) and isinstance(right, (int, float)):
                return left * right
            if isinstance(left, str) and isinstance(right, int):
                return left * right
            if isinstance(left, list) and isinstance(right, int):
                return left * right
            return error(f'cannot multiply {type(left).__name__} and {type(right).__name__}')
        if node.op == '/':
            if isinstance(left, (int, float)) and isinstance(right, (int, float)):
                if right == 0:
                    return error('division by zero')
                return left / right
            return error(f'cannot divide {type(left).__name__} by {type(right).__name__}')
        if node.op == '%':
            if isinstance(left, int) and isinstance(right, int):
                if right == 0:
                    return error('modulo by zero')
                return left % right
            return error(f'cannot mod {type(left).__name__} by {type(right).__name__}')
        if node.op == '==':
            return left == right
        if node.op == '!=':
            return left != right
        if node.op == '<':
            if isinstance(left, (int, float)) and isinstance(right, (int, float)):
                return left < right
            return error(f'cannot compare {type(left).__name__} and {type(right).__name__}')
        if node.op == '>':
            if isinstance(left, (int, float)) and isinstance(right, (int, float)):
                return left > right
            return error(f'cannot compare {type(left).__name__} and {type(right).__name__}')
        if node.op == '<=':
            if isinstance(left, (int, float)) and isinstance(right, (int, float)):
                return left <= right
            return error(f'cannot compare {type(left).__name__} and {type(right).__name__}')
        if node.op == '>=':
            if isinstance(left, (int, float)) and isinstance(right, (int, float)):
                return left >= right
            return error(f'cannot compare {type(left).__name__} and {type(right).__name__}')
        if node.op == '&&':
            if not isinstance(left, bool):
                return error(f'expected bool, got {type(left).__name__}')
            if not left:
                return False
            if not isinstance(right, bool):
                return error(f'expected bool, got {type(right).__name__}')
            return right
        if node.op == '||':
            if not isinstance(left, bool):
                return error(f'expected bool, got {type(left).__name__}')
            if left:
                return True
            if not isinstance(right, bool):
                return error(f'expected bool, got {type(right).__name__}')
            return right
        return error(f'unknown operator: {node.op}')
    
    if isinstance(node, UnaryOp):
        operand = eval_node(node.operand, env)
        if is_error(operand):
            return operand
        if node.op == '-':
            if isinstance(operand, (int, float)):
                return -operand
            return error(f'cannot negate {type(operand).__name__}')
        if node.op == '!':
            if isinstance(operand, bool):
                return not operand
            return error(f'expected bool, got {type(operand).__name__}')
        return error(f'unknown unary op: {node.op}')
    
    if isinstance(node, Call):
        func = eval_node(node.func, env)
        if is_error(func):
            return func
        
        args = []
        for arg_node in node.args:
            arg = eval_node(arg_node, env)
            if is_error(arg):
                return arg
            args.append(arg)
        
        if isinstance(func, Closure):
            if len(args) != len(func.params):
                return error(f'expected {len(func.params)} args, got {len(args)}')
            call_env = func.env.extend()
            for param, arg in zip(func.params, args):
                call_env.define(param, arg)
            return eval_node(func.body, call_env)
        
        if callable(func):
            try:
                return func(*args)
            except Exception as e:
                return error(str(e))
        
        return error(f'cannot call {type(func).__name__}')
    
    if isinstance(node, If):
        cond = eval_node(node.cond, env)
        if is_error(cond):
            return cond
        if not isinstance(cond, bool):
            return error(f'condition must be bool, got {type(cond).__name__}')
        if cond:
            return eval_node(node.then_expr, env)
        else:
            return eval_node(node.else_expr, env)
    
    if isinstance(node, Fn):
        return Closure(node.params, node.body, env)
    
    if isinstance(node, Let):
        value = eval_node(node.value, env)
        if is_error(value):
            return value
        env.define(node.name, value)
        return None
    
    if isinstance(node, Program):
        for stmt in node.stmts:
            result = eval_node(stmt, env)
            if is_error(result):
                return result
        return None
    
    return error(f'unknown node type: {type(node).__name__}')

def eval_program(src: str) -> Any:
    try:
        ast = parse(src)
        env = Env()
        env.define('print', lambda *args: (print(' '.join(str(a) for a in args)), None)[1])
        env.define('len', lambda x: len(x) if isinstance(x, (list, str, dict)) else error(f'len() of {type(x).__name__}'))
        result = eval_node(ast, env)
        return result
    except LexError as e:
        return error(str(e))
    except ParseError as e:
        return error(str(e))
```

```python
#!/usr/bin/env python3
import sys
import os
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from expr.evaluator import eval_program, is_error

def main():
    if len(sys.argv) != 2:
        print("usage: run.py <file.expr>", file=sys.stderr)
        return 2
    
    filepath = sys.argv[1]
    try:
        with open(filepath) as f:
            src = f.read()
    except FileNotFoundError:
        print(f"error: file not found: {filepath}", file=sys.stderr)
        return 2
    except Exception as e:
        print(f"error: {e}", file=sys.stderr)
        return 2
    
    result = eval_program(src)
    if is_error(result):
        print(result['error'], file=sys.stderr)
        return 1
    
    return 0

if __name__ == '__main__':
    sys.exit(main())
```

```python
import pytest
from expr.lexer import tokenize, LexError

def test_numbers():
    tokens = tokenize("42 3.14;")

def test_operators():
    tokens = tokenize("+ - * / % == != < > <= >= && ||")
    ops = [t.value for t in tokens if t.type not in ['EOF']]
    assert ops == ['+', '-', '*', '/', '%', '==', '!=', '<', '>', '<=', '>=', '&&', '||']

def test_names():
    tokens = tokenize("foo bar_baz _var")
    assert tokens[0].value == "foo"
    assert tokens[1].value == "bar_baz"
    assert tokens[2].value == "_var"

def test_unterminated_string():
    with pytest.raises(LexError) as exc:
        tokenize('"unclosed')
    assert "unterminated string" in str(exc.value)

def test_line_col_tracking():
    tokens = tokenize("42\nfoo")
    assert tokens[0].line == 1 and tokens[0].col == 1
    assert tokens[1].line == 2 and tokens[1].col == 1
```
