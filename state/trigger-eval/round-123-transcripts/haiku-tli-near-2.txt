```markdown
# Spark Language Specification

## Overview
Spark is a small functional scripting language with first-class functions, lists, records, and error handling as propagating values.

## Core Idea
Errors are values, not exceptions. All runtime faults (unbound names, division by zero, arity mismatch) produce a `Miss` value that propagates through the program. This makes the evaluator total—no Python exceptions for user errors.

## Design Decisions
- **Errors-as-values:** Every runtime fault becomes a Miss with a reason string
- **Newline-sensitive:** Statements end at newlines; brackets suppress them
- **Immutable data:** Lists and records are immutable
- **Total evaluator:** RecursionError caught at call boundary → Miss value

## Grammar (Precedence, low→high)
```
Program    → Statement* EOF
Statement  → Let | Expr
Let        → "let" ID "=" Expr
Expr       → IfExpr | OrExpr
IfExpr     → "if" Expr "then" Expr "else" Expr
OrExpr     → AndExpr ("||" AndExpr)*
AndExpr    → CmpExpr ("&&" CmpExpr)*
CmpExpr    → AddExpr ((== | != | < | > | <= | >=) AddExpr)*
AddExpr    → MulExpr ((+ | -) MulExpr)*
MulExpr    → UnaryExpr ((* | /) UnaryExpr)*
UnaryExpr  → (- | not) UnaryExpr | CallExpr
CallExpr   → PrimaryExpr (("(" ArgList? ")") | ("[" Expr "]") | ("." ID))*
PrimaryExpr→ NUM | STR | BOOL | ID | "(" Expr ")" | "[" List "]" | "{" Record "}" | FnExpr | "missed" "(" Expr ")"
FnExpr     → "fn" "(" Params? ")" Expr
List       → (Expr ("," Expr)*)? ","?
Record     → (ID ":" Expr ("," ID ":" Expr)*)? ","?
ArgList    → Expr ("," Expr)*
```

## Exit Codes
- 0: Program succeeds, output result
- 1: Program returns a Miss value (error)
- 2: Lex or parse error (clean error message with line:col)

## Built-in Functions
- `len(x)` — list/record/string length
- `map(fn, list)` — apply fn to each element
- `filter(fn, list)` — keep elements where fn returns true
- `fold(fn, init, list)` — reduce with fn
- `print(...)` — output to stdout (returns nil)
- `type(x)` — return type name as string
```

```python
class Miss:
    def __init__(self, reason: str):
        self.reason = reason
    def __repr__(self):
        return f"Miss({self.reason})"
    def __eq__(self, other):
        return isinstance(other, Miss) and self.reason == other.reason

def missed(x):
    return isinstance(x, Miss)

@dataclass
class Token:
    type: str
    value: Any
    line: int
    col: int

def tokenize(src: str) -> List[Token]:
    tokens = []
    line, col = 1, 1
    i = 0
    bracket_depth = 0
    
    while i < len(src):
        if src[i] == '\n':
            line += 1
            col = 1
            i += 1
            if bracket_depth == 0:
                while i < len(src) and src[i] == '\n':
                    line += 1
                    i += 1
                if i < len(src):
                    tokens.append(Token('NEWLINE', '\n', line-1, col))

        if src[i].isspace():
            col += 1
            i += 1
            continue
        
        if src[i:i+2] == '//':

            i += 2
            col += 2
            continue
        
        if src[i] in '()[]{},.':

            if char in '([':

            elif char in ')]':

            col += 1
            continue

            col += 1

            col += 1
            s = ''

                if src[i] == '\\' and i + 1 < len(src):
                    s += src[i+1]
                    i += 2

                    s += src[i]
                    i += 1
                    col += 1
            if i >= len(src):
                raise SyntaxError(f"Unterminated string at {line}:{start_col}")
            tokens.append(Token('STR', s, line, start_col))
            i += 1
            col += 1
            continue
        
        if src[i].isdigit() or (src[i] == '-' and i+1 < len(src) and src[i+1].isdigit()):
            j = i
            if src[i] == '-':
                j += 1
            while j < len(src) and (src[j].isdigit() or src[j] == '.'):
                j += 1
            num_str = src[i:j]
            if '.' in num_str:
                try:
                    val = float(num_str)
                except (ValueError, OverflowError):
                    raise SyntaxError(f"Invalid number at {line}:{start_col}")
            else:
                val = int(num_str)
            tokens.append(Token('NUM', val, line, start_col))
            col += j - i
            i = j
            continue
        
        if src[i].isalpha() or src[i] == '_':
            j = i
            while j < len(src) and (src[j].isalnum() or src[j] == '_'):
                j += 1
            word = src[i:j]
            if word in ('true', 'false'):
                tokens.append(Token('BOOL', word == 'true', line, start_col))
            elif word in ('let', 'if', 'then', 'else', 'fn', 'missed'):
                tokens.append(Token(word.upper(), word, line, start_col))
            else:
                tokens.append(Token('ID', word, line, start_col))
            col += j - i
            i = j
            continue
        
        raise SyntaxError(f"Unexpected character '{src[i]}' at {line}:{start_col}")
    
    tokens.append(Token('EOF', None, line, col))
    return tokens

def _simple(name, fields):
    def __init__(self, **kwargs):
        for field in fields:
            setattr(self, field, kwargs.get(field))
        self.line = kwargs.get('line', 1)
    
    cls = type(name, (), {'__init__': __init__})
    return cls

Num = _simple('Num', ['value'])
Str = _simple('Str', ['value'])
Bool = _simple('Bool', ['value'])
List = _simple('List', ['items'])
Record = _simple('Record', ['fields'])
Id = _simple('Id', ['name'])
BinOp = _simple('BinOp', ['op', 'left', 'right'])
UnaryOp = _simple('UnaryOp', ['op', 'expr'])
Call = _simple('Call', ['fn', 'args'])
Index = _simple('Index', ['expr', 'index'])
Field = _simple('Field', ['expr', 'name'])
IfExpr = _simple('IfExpr', ['cond', 'then_expr', 'else_expr'])
Let = _simple('Let', ['name', 'value', 'body'])
Fn = _simple('Fn', ['params', 'body'])
Missed = _simple('Missed', ['expr'])

class Parser:
    def __init__(self, tokens):
        self.tokens = tokens
        self.pos = 0
    
    def current(self):
        return self.tokens[self.pos] if self.pos < len(self.tokens) else self.tokens[-1]
    
    def peek(self, offset=1):
        pos = self.pos + offset
        return self.tokens[pos] if pos < len(self.tokens) else self.tokens[-1]
    
    def advance(self):
        self.pos += 1
    
    def expect(self, ttype):
        if self.current().type != ttype:
            raise SyntaxError(f"Expected {ttype}, got {self.current().type} at {self.current().line}:{self.current().col}")
        tok = self.current()
        self.advance()
        return tok
    
    def skip_newlines(self):
        while self.current().type == 'NEWLINE':
            self.advance()
    
    def parse(self):
        self.skip_newlines()
        statements = []
        while self.current().type != 'EOF':
            if self.current().type == 'LET':
                statements.append(self.parse_let())
            else:
                statements.append(self.parse_expr())
            self.skip_newlines()
        
        if not statements:
            statements.append(Num(value=0, line=1))
        
        result = statements[-1]
        for stmt in reversed(statements[:-1]):
            result = Let(name=stmt.name, value=stmt.value, body=result, line=stmt.line)
        
        return result
    
    def parse_let(self):
        tok = self.expect('LET')
        name_tok = self.expect('ID')
        self.expect('=')
        value = self.parse_expr()
        return Let(name=name_tok.value, value=value, body=Num(value=0, line=tok.line), line=tok.line)
    
    def parse_expr(self):
        return self.parse_if()
    
    def parse_if(self):
        if self.current().type == 'IF':
            tok = self.current()
            self.advance()
            cond = self.parse_or()
            self.expect('THEN')
            then_expr = self.parse_or()
            self.expect('ELSE')
            else_expr = self.parse_or()
            return IfExpr(cond=cond, then_expr=then_expr, else_expr=else_expr, line=tok.line)
        return self.parse_or()
    
    def parse_or(self):
        left = self.parse_and()
        while self.current().type == '||':
            tok = self.current()
            self.advance()
            right = self.parse_and()
            left = BinOp(op='||', left=left, right=right, line=tok.line)
        return left
    
    def parse_and(self):
        left = self.parse_cmp()
        while self.current().type == '&&':
            tok = self.current()
            self.advance()
            right = self.parse_cmp()
            left = BinOp(op='&&', left=left, right=right, line=tok.line)
        return left
    
    def parse_cmp(self):
        left = self.parse_add()
        while self.current().type in ('==', '!=', '<', '>', '<=', '>='):
            tok = self.current()
            self.advance()
            right = self.parse_add()
            left = BinOp(op=tok.type, left=left, right=right, line=tok.line)
        return left
    
    def parse_add(self):
        left = self.parse_mul()
        while self.current().type in ('+', '-'):
            tok = self.current()
            self.advance()
            right = self.parse_mul()
            left = BinOp(op=tok.value, left=left, right=right, line=tok.line)
        return left
    
    def parse_mul(self):
        left = self.parse_unary()
        while self.current().type in ('*', '/'):
            tok = self.current()
            self.advance()
            right = self.parse_unary()
            left = BinOp(op=tok.value, left=left, right=right, line=tok.line)
        return left
    
    def parse_unary(self):
        if self.current().type in ('-', 'not'):
            tok = self.current()
            self.advance()
            expr = self.parse_unary()
            return UnaryOp(op=tok.value, expr=expr, line=tok.line)
        return self.parse_call()
    
    def parse_call(self):
        expr = self.parse_primary()
        while True:
            if self.current().type == '(':
                self.advance()
                args = []
                if self.current().type != ')':
                    args.append(self.parse_expr())
                    while self.current().type == ',':
                        self.advance()
                        args.append(self.parse_expr())
                self.expect(')')
                expr = Call(fn=expr, args=args, line=expr.line)
            elif self.current().type == '[':
                self.advance()
                index = self.parse_expr()
                self.expect(']')
                expr = Index(expr=expr, index=index, line=expr.line)
            elif self.current().type == '.':
                self.advance()
                name = self.expect('ID').value
                expr = Field(expr=expr, name=name, line=expr.line)
            else:
                break
        return expr
    
    def parse_primary(self):
        tok = self.current()
        
        if tok.type == 'NUM':
            self.advance()
            return Num(value=tok.value, line=tok.line)
        
        if tok.type == 'STR':
            self.advance()
            return Str(value=tok.value, line=tok.line)
        
        if tok.type == 'BOOL':
            self.advance()
            return Bool(value=tok.value, line=tok.line)
        
        if tok.type == 'ID':
            self.advance()
            return Id(name=tok.value, line=tok.line)
        
        if tok.type == '(':
            self.advance()
            expr = self.parse_expr()
            self.expect(')')
            return expr
        
        if tok.type == '[':
            self.advance()
            items = []
            if self.current().type != ']':
                items.append(self.parse_expr())
                while self.current().type == ',':
                    self.advance()
                    if self.current().type == ']':
                        break
                    items.append(self.parse_expr())
            self.expect(']')
            return List(items=items, line=tok.line)
        
        if tok.type == '{':
            self.advance()
            fields = {}
            if self.current().type != '}':
                name = self.expect('ID').value
                self.expect(':')
                value = self.parse_expr()
                fields[name] = value
                while self.current().type == ',':
                    self.advance()
                    if self.current().type == '}':
                        break
                    name = self.expect('ID').value
                    self.expect(':')
                    value = self.parse_expr()
                    fields[name] = value
            self.expect('}')
            return Record(fields=fields, line=tok.line)
        
        if tok.type == 'FN':
            self.advance()
            self.expect('(')
            params = []
            if self.current().type != ')':
                params.append(self.expect('ID').value)
                while self.current().type == ',':
                    self.advance()
                    params.append(self.expect('ID').value)
            self.expect(')')
            body = self.parse_expr()
            return Fn(params=params, body=body, line=tok.line)
        
        if tok.type == 'MISSED':
            self.advance()
            self.expect('(')
            expr = self.parse_expr()
            self.expect(')')
            return Missed(expr=expr, line=tok.line)
        
        raise SyntaxError(f"Unexpected token {tok.type} at {tok.line}:{tok.col}")

class Env:
    def __init__(self, parent=None):
        self.parent = parent
        self.bindings = {}
    
    def define(self, name, value):
        self.bindings[name] = value
    
    def get(self, name):
        if name in self.bindings:
            return self.bindings[name]
        if self.parent:
            return self.parent.get(name)
        return Miss(f"unbound name: {name}")

class Evaluator:
    def __init__(self, max_depth=1000):
        self.max_depth = max_depth
        self.depth = 0
    
    def eval(self, node, env):
        self.depth += 1
        if self.depth > self.max_depth:
            self.depth -= 1
            return Miss("max recursion depth exceeded")
        
        try:
            if isinstance(node, Num):
                return node.value
            elif isinstance(node, Str):
                return node.value
            elif isinstance(node, Bool):
                return node.value
            elif isinstance(node, List):
                items = []
                for item in node.items:
                    val = self.eval(item, env)
                    items.append(val)
                return items
            elif isinstance(node, Record):
                fields = {}
                for name, expr in node.fields.items():
                    val = self.eval(expr, env)
                    fields[name] = val
                return fields
            elif isinstance(node, Id):
                return env.get(node.name)
            elif isinstance(node, BinOp):
                return self.eval_binop(node, env)
            elif isinstance(node, UnaryOp):
                return self.eval_unary(node, env)
            elif isinstance(node, Call):
                return self.eval_call(node, env)
            elif isinstance(node, Index):
                return self.eval_index(node, env)
            elif isinstance(node, Field):
                return self.eval_field(node, env)
            elif isinstance(node, IfExpr):
                cond = self.eval(node.cond, env)
                if missed(cond):
                    return cond
                if not isinstance(cond, bool):
                    return Miss(f"if condition must be bool, got {type(cond).__name__}")
                if cond:
                    return self.eval(node.then_expr, env)
                else:
                    return self.eval(node.else_expr, env)
            elif isinstance(node, Let):
                value = self.eval(node.value, env)
                if missed(value):
                    return value
                new_env = Env(env)
                new_env.define(node.name, value)
                return self.eval(node.body, new_env)
            elif isinstance(node, Fn):
                return ('fn', node.params, node.body, env)
            elif isinstance(node, Missed):
                val = self.eval(node.expr, env)
                return isinstance(val, Miss)
            else:
                return Miss(f"unknown node type: {type(node).__name__}")
        finally:
            self.depth -= 1
    
    def eval_binop(self, node, env):
        left = self.eval(node.left, env)
        if missed(left):
            return left
        right = self.eval(node.right, env)
        if missed(right):
            return right
        
        if node.op == '+':
            if isinstance(left, (int, float)) and isinstance(right, (int, float)):
                return left + right
            if isinstance(left, str) and isinstance(right, str):
                return left + right
            if isinstance(left, list) and isinstance(right, list):
                return left + right
            return Miss(f"cannot add {type(left).__name__} and {type(right).__name__}")
        elif node.op == '-':
            if isinstance(left, (int, float)) and isinstance(right, (int, float)):
                return left - right
            return Miss(f"cannot subtract {type(right).__name__} from {type(left).__name__}")
        elif node.op == '*':
            if isinstance(left, (int, float)) and isinstance(right, (int, float)):
                return left * right
            if isinstance(left, str) and isinstance(right, int):
                return left * right
            if isinstance(left, list) and isinstance(right, int):
                return left * right
            return Miss(f"cannot multiply {type(left).__name__} and {type(right).__name__}")
        elif node.op == '/':
            if isinstance(left, (int, float)) and isinstance(right, (int, float)):
                if right == 0:
                    return Miss("division by zero")
                return left / right
            return Miss(f"cannot divide {type(left).__name__} by {type(right).__name__}")
        elif node.op == '==':
            return left == right
        elif node.op == '!=':
            return left != right
        elif node.op == '<':
            if isinstance(left, (int, float)) and isinstance(right, (int, float)):
                return left < right
            return Miss(f"cannot compare {type(left).__name__} and {type(right).__name__}")
        elif node.op == '>':
            if isinstance(left, (int, float)) and isinstance(right, (int, float)):
                return left > right
            return Miss(f"cannot compare {type(left).__name__} and {type(right).__name__}")
        elif node.op == '<=':
            if isinstance(left, (int, float)) and isinstance(right, (int, float)):
                return left <= right
            return Miss(f"cannot compare {type(left).__name__} and {type(right).__name__}")
        elif node.op == '>=':
            if isinstance(left, (int, float)) and isinstance(right, (int, float)):
                return left >= right
            return Miss(f"cannot compare {type(left).__name__} and {type(right).__name__}")
        elif node.op == '||':
            if not isinstance(left, bool):
                return Miss(f"|| requires bool, got {type(left).__name__}")
            if not isinstance(right, bool):
                return Miss(f"|| requires bool, got {type(right).__name__}")
            return left or right
        elif node.op == '&&':
            if not isinstance(left, bool):
                return Miss(f"&& requires bool, got {type(left).__name__}")
            if not isinstance(right, bool):
                return Miss(f"&& requires bool, got {type(right).__name__}")
            return left and right
        
        return Miss(f"unknown operator: {node.op}")
    
    def eval_unary(self, node, env):
        expr = self.eval(node.expr, env)
        if missed(expr):
            return expr
        
        if node.op == '-':
            if isinstance(expr, (int, float)):
                return -expr
            return Miss(f"cannot negate {type(expr).__name__}")
        elif node.op == 'not':
            if isinstance(expr, bool):
                return not expr
            return Miss(f"not requires bool, got {type(expr).__name__}")
        
        return Miss(f"unknown unary operator: {node.op}")
    
    def eval_call(self, node, env):
        fn = self.eval(node.fn, env)
        if missed(fn):
            return fn
        
        if callable(fn):
            args = []
            for arg_node in node.args:
                arg = self.eval(arg_node, env)
                args.append(arg)
            return fn(*args)
        
        if isinstance(fn, tuple) and fn[0] == 'fn':
            _, params, body, closure_env = fn
            if len(params) != len(node.args):
                return Miss(f"function expects {len(params)} args, got {len(node.args)}")
            
            args = []
            for arg_node in node.args:
                arg = self.eval(arg_node, env)
                args.append(arg)
            
            new_env = Env(closure_env)
            for param, arg in zip(params, args):
                new_env.define(param, arg)
            
            return self.eval(body, new_env)
        
        return Miss(f"cannot call {type(fn).__name__}")
    
    def eval_index(self, node, env):
        expr = self.eval(node.expr, env)
        if missed(expr):
            return expr
        
        index = self.eval(node.index, env)
        if missed(index):
            return index
        
        if isinstance(expr, list) and isinstance(index, int):
            if -len(expr) <= index < len(expr):
                return expr[index]
            return Miss(f"list index out of range: {index}")
        elif isinstance(expr, dict) and isinstance(index, str):
            return expr.get(index, Miss(f"key not found: {index}"))
        
        return Miss(f"cannot index {type(expr).__name__} with {type(index).__name__}")
    
    def eval_field(self, node, env):
        expr = self.eval(node.expr, env)
        if missed(expr):
            return expr
        
        if isinstance(expr, dict) and node.name in expr:
            return expr[node.name]
        
        return Miss(f"field not found: {node.name}")

def repr_value(val):
    if isinstance(val, bool):
        return 'true' if val else 'false'
    elif isinstance(val, str):
        return repr(val)
    elif isinstance(val, list):
        return '[' + ', '.join(repr_value(v) for v in val) + ']'
    elif isinstance(val, dict):
        items = ', '.join(f'{k}: {repr_value(v)}' for k, v in val.items())
        return '{' + items + '}'
    elif isinstance(val, Miss):
        return f'error: {val.reason}'
    else:
        return str(val)

def run(src):
    try:
        tokens = tokenize(src)
        parser = Parser(tokens)
        ast = parser.parse()
        evaluator = Evaluator()
        env = Env()
        
        result = evaluator.eval(ast, env)
        
        if isinstance(result, Miss):
            return 1, result.reason
        else:
            return 0, repr_value(result)
    except SyntaxError as e:
        return 2, str(e)
    except Exception as e:
        return 2, f"Error: {str(e)}"
```

```python
            src = f.read()
    except IOError as e:
        print(f"Error reading file: {e}", file=sys.stderr)
        sys.exit(2)
    
    exit_code, output = run(src)
    if output:
        print(output)
```
