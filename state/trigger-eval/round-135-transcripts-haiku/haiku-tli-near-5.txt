spec = '''# Spark Language Specification

## Overview
Spark is a minimal dynamically-typed scripting language with functions, lists, and error propagation as first-class values.

## Core Ideas
- **Errors as values**: Every operation that can fail produces a `Miss(reason)` value instead of raising. This value propagates until rescued or printed.
- **Total evaluator**: No exceptions escape to Python. Runtime faults (unbound name, bad index, div-by-zero) become catchable errors.
- **Simple semantics**: Numbers, strings, lists, functions, conditionals, loops. No classes, no modules.

## Grammar (precedence low→high)

```
program       → (stmt)*
stmt          → (let_stmt | if_stmt | return_stmt | expr) NEWLINE*
let_stmt      → "let" NAME "=" expr
if_stmt       → "if" expr "{" block "}" ("else" "{" block "}")?
return_stmt   → "return" expr?
block         → (stmt)* expr?

expr          → or_expr
or_expr       → and_expr ("||" and_expr)*
and_expr      → eq_expr ("&&" eq_expr)*
eq_expr       → cmp_expr (("==" | "!=") cmp_expr)*
cmp_expr      → add_expr (("<" | "<=" | ">" | ">=") add_expr)*
add_expr      → mul_expr (("+" | "-") mul_expr)*
mul_expr      → unary_expr (("*" | "/" | "%") unary_expr)*
unary_expr    → ("!" | "-") unary_expr | call_expr
call_expr     → primary_expr ("(" args? ")")*
primary_expr  → NUMBER | STRING | NAME | list_expr | fn_expr | "(" expr ")" | rescue_expr

list_expr     → "[" (expr ("," expr)*)? "]"
fn_expr       → "fn" "(" params? ")" "{" block "}"
params        → NAME ("," NAME)*
args          → expr ("," expr)*
rescue_expr   → "rescue" expr
```

## Exit Codes
- `0`: program completed successfully
- `1`: program ran and hit a runtime error (Miss value not rescued)
- `2`: lexer or parser error

## Semantics
- **Variables**: `let x = 5; x + 1` → 6
- **Functions**: `let f = fn(x) { x + 1 }; f(5)` → 6
- **Lists**: `[1, 2, 3]`, `xs[0]` → index access
- **Conditionals**: `if x { 1 } else { 2 }`
- **Errors**: `1 / 0` → `Miss("division by zero")`, `rescue` catches them
- **Types**: number, string, boolean, list, function, Miss

## Operators
- Arithmetic: `+`, `-`, `*`, `/`, `%`
- Comparison: `==`, `!=`, `<`, `<=`, `>`, `>=`
- Logical: `&&`, `||`, `!`
- Index: `list[n]`, string index
- Rescue: `rescue expr` → value or 0 if Miss

## Built-in Functions
- `print(x)` → prints to stdout
- `len(x)` → length of list/string
- `type(x)` → string name of type
- `missed(x)` → true if x is a Miss value
- `push(list, item)` → new list with item appended
'''

parser_code = '''"""Parser for Spark language."""
from dataclasses import dataclass
from typing import List, Optional, Union, Any
from lexer import Token, tokenize, LexError

class ParseError(Exception):
    def __init__(self, msg: str, token: Optional[Token] = None):
        self.msg = msg
        self.token = token
    
    def __str__(self):
        if self.token:
            return f"error: {self.msg} at line {self.token.line}, col {self.token.col}"
        return f"error: {self.msg}"

def _simple(name: str, fields: str):
    """Factory for simple AST nodes."""
    class Node:
        def __init__(self, *args, line=0):
            self.line = line
            for f, v in zip(fields.split(), args):
                setattr(self, f, v)
        def __repr__(self):
            attrs = ', '.join(f"{f}={getattr(self, f)!r}" for f in fields.split())
            return f"{name}({attrs})"
    Node.__name__ = name
    return Node

# AST node types
Number = _simple('Number', 'value')
String = _simple('String', 'value')
Boolean = _simple('Boolean', 'value')
Name = _simple('Name', 'name')
List = _simple('List', 'elements')
Index = _simple('Index', 'obj index')
BinOp = _simple('BinOp', 'op left right')
UnaryOp = _simple('UnaryOp', 'op operand')
Call = _simple('Call', 'fn args')
FnDef = _simple('FnDef', 'params body')
Let = _simple('Let', 'name value')
If = _simple('If', 'cond then_block else_block')
Return = _simple('Return', 'value')
Block = _simple('Block', 'stmts')
Rescue = _simple('Rescue', 'expr')

class Parser:
    def __init__(self, tokens: List[Token]):
        self.tokens = tokens
        self.pos = 0
    
    def current(self) -> Token:
        if self.pos < len(self.tokens):
            return self.tokens[self.pos]
        return self.tokens[-1]  # EOF
    
    def peek(self, offset: int = 1) -> Token:
        pos = self.pos + offset
        if pos < len(self.tokens):
            return self.tokens[pos]
        return self.tokens[-1]
    
    def advance(self) -> Token:
        tok = self.current()
        if tok.type != 'EOF':
            self.pos += 1
        return tok
    
    def expect(self, typ: str) -> Token:
        tok = self.current()
        if tok.type != typ:
            raise ParseError(f"expected {typ}, got {tok.type}", tok)
        return self.advance()
    
    def match(self, *types: str) -> bool:
        return self.current().type in types
    
    def parse(self) -> Block:
        stmts = []
        while not self.match('EOF'):
            stmt = self.parse_stmt()
            stmts.append(stmt)
            self.skip_newlines()
        return Block(stmts, line=1)
    
    def skip_newlines(self):
        # In this simple version, we don't emit NEWLINE tokens
        pass
    
    def parse_stmt(self):
        if self.match('LET'):
            return self.parse_let()
        elif self.match('IF'):
            return self.parse_if()
        elif self.match('RETURN'):
            return self.parse_return()
        else:
            return self.parse_expr()
    
    def parse_let(self):
        line = self.current().line
        self.expect('LET')
        name_tok = self.expect('NAME')
        self.expect('=')
        value = self.parse_expr()
        return Let(name_tok.value, value, line=line)
    
    def parse_if(self):
        line = self.current().line
        self.expect('IF')
        cond = self.parse_expr()
        self.expect('{')
        then_block = self.parse_block()
        self.expect('}')
        else_block = None
        if self.match('ELSE'):
            self.advance()
            self.expect('{')
            else_block = self.parse_block()
            self.expect('}')
        return If(cond, then_block, else_block, line=line)
    
    def parse_return(self):
        line = self.current().line
        self.expect('RETURN')
        value = None
        if not self.match('}', 'EOF'):
            value = self.parse_expr()
        return Return(value, line=line)
    
    def parse_block(self):
        stmts = []
        while not self.match('}', 'EOF'):
            stmt = self.parse_stmt()
            stmts.append(stmt)
        # Last stmt in block is the value
        return Block(stmts, line=self.current().line)
    
    def parse_expr(self):
        return self.parse_or()
    
    def parse_or(self):
        left = self.parse_and()
        while self.match('||'):
            op_tok = self.advance()
            right = self.parse_and()
            left = BinOp(op_tok.value, left, right, line=op_tok.line)
        return left
    
    def parse_and(self):
        left = self.parse_eq()
        while self.match('&&'):
            op_tok = self.advance()
            right = self.parse_eq()
            left = BinOp(op_tok.value, left, right, line=op_tok.line)
        return left
    
    def parse_eq(self):
        left = self.parse_cmp()
        while self.match('==', '!='):
            op_tok = self.advance()
            right = self.parse_cmp()
            left = BinOp(op_tok.value, left, right, line=op_tok.line)
        return left
    
    def parse_cmp(self):
        left = self.parse_add()
        while self.match('<', '<=', '>', '>='):
            op_tok = self.advance()
            right = self.parse_add()
            left = BinOp(op_tok.value, left, right, line=op_tok.line)
        return left
    
    def parse_add(self):
        left = self.parse_mul()
        while self.match('+', '-'):
            op_tok = self.advance()
            right = self.parse_mul()
            left = BinOp(op_tok.value, left, right, line=op_tok.line)
        return left
    
    def parse_mul(self):
        left = self.parse_unary()
        while self.match('*', '/', '%'):
            op_tok = self.advance()
            right = self.parse_unary()
            left = BinOp(op_tok.value, left, right, line=op_tok.line)
        return left
    
    def parse_unary(self):
        if self.match('!', '-'):
            op_tok = self.advance()
            operand = self.parse_unary()
            return UnaryOp(op_tok.value, operand, line=op_tok.line)
        return self.parse_call()
    
    def parse_call(self):
        expr = self.parse_primary()
        while self.match('('):
            self.advance()
            args = []
            if not self.match(')'):
                args.append(self.parse_expr())
                while self.match(','):
                    self.advance()
                    args.append(self.parse_expr())
            self.expect(')')
            expr = Call(expr, args, line=expr.line)
        return expr
    
    def parse_primary(self):
        if self.match('NUMBER'):
            tok = self.advance()
            return Number(tok.value, line=tok.line)
        
        if self.match('STRING'):
            tok = self.advance()
            return String(tok.value, line=tok.line)
        
        if self.match('TRUE', 'FALSE'):
            tok = self.advance()
            return Boolean(tok.type == 'TRUE', line=tok.line)
        
        if self.match('NAME'):
            name_tok = self.advance()
            # Check for index access
            if self.match('['):
                self.advance()
                index = self.parse_expr()
                self.expect(']')
                return Index(Name(name_tok.value, line=name_tok.line), index, line=name_tok.line)
            return Name(name_tok.value, line=name_tok.line)
        
        if self.match('['):
            return self.parse_list()
        
        if self.match('FN'):
            return self.parse_fn()
        
        if self.match('('):
            self.advance()
            expr = self.parse_expr()
            self.expect(')')
            return expr
        
        if self.match('RESCUE'):
            line = self.current().line
            self.advance()
            expr = self.parse_expr()
            return Rescue(expr, line=line)
        
        raise ParseError(f"unexpected token {self.current().type}", self.current())
    
    def parse_list(self):
        line = self.current().line
        self.expect('[')
        elements = []
        if not self.match(']'):
            elements.append(self.parse_expr())
            while self.match(','):
                self.advance()
                if self.match(']'):
                    break
                elements.append(self.parse_expr())
        self.expect(']')
        return List(elements, line=line)
    
    def parse_fn(self):
        line = self.current().line
        self.expect('FN')
        self.expect('(')
        params = []
        if not self.match(')'):
            params.append(self.expect('NAME').value)
            while self.match(','):
                self.advance()
                params.append(self.expect('NAME').value)
        self.expect(')')
        self.expect('{')
        body = self.parse_block()
        self.expect('}')
        return FnDef(params, body, line=line)

def parse(src: str):
    """Parse Spark source code. Returns AST."""
    try:
        tokens = tokenize(src)
        parser = Parser(tokens)
        return parser.parse()
    except LexError as e:
        raise ParseError(str(e))
'''

from dataclasses import dataclass
from typing import Any, Dict, List, Optional, Callable
from parser import (
    Number, String, Boolean, Name, List as ListNode, Index, BinOp, UnaryOp,

)

@dataclass
class Miss:
    """Error value that propagates."""
    reason: str
    line: int = 0

def missed(val: Any) -> bool:
    """Check if value is an error."""
    return isinstance(val, Miss)

class Env:
    """Environment for variable bindings."""
    def __init__(self, parent: Optional['Env'] = None):
        self.parent = parent
        self.bindings: Dict[str, Any] = {}
    
    def define(self, name: str, value: Any) -> None:
        self.bindings[name] = value
    
    def get(self, name: str) -> Any:
        if name in self.bindings:
            return self.bindings[name]
        if self.parent:
            return self.parent.get(name)
        return Miss(f"unbound name '{name}'")

class SparkFunction:
    """User-defined function."""
    def __init__(self, params: List[str], body: Any, closure: Env):
        self.params = params
        self.body = body
        self.closure = closure
    
    def __repr__(self):
        return f"<fn({len(self.params)} params)>"

class Evaluator:
    def __init__(self):
        self.builtins = self._make_builtins()
    
    def _make_builtins(self) -> Dict[str, Callable]:
        return {
            'print': self._builtin_print,
            'len': self._builtin_len,
            'type': self._builtin_type,
            'missed': self._builtin_missed,
            'push': self._builtin_push,
        }
    
    def _builtin_print(self, x: Any) -> Any:
        print(self._str(x))
        return None
    
    def _builtin_len(self, x: Any) -> Any:
        if isinstance(x, list):
            return len(x)
        if isinstance(x, str):
            return len(x)
        return Miss("len() requires list or string")
    
    def _builtin_type(self, x: Any) -> Any:
        if isinstance(x, bool):
            return "bool"
        if isinstance(x, int) or isinstance(x, float):
            return "number"
        if isinstance(x, str):
            return "string"
        if isinstance(x, list):
            return "list"
        if isinstance(x, SparkFunction):
            return "function"
        if isinstance(x, Miss):
            return "miss"
        return "unknown"
    
    def _builtin_missed(self, x: Any) -> Any:
        return isinstance(x, Miss)
    
    def _builtin_push(self, lst: Any, item: Any) -> Any:
        if not isinstance(lst, list):
            return Miss("push() requires a list")
        return lst + [item]
    
    def _str(self, val: Any) -> str:
        if isinstance(val, bool):
            return "true" if val else "false"
        if isinstance(val, Miss):
            return f"Miss({val.reason})"
        if val is None:
            return "null"
        if isinstance(val, list):
            return "[" + ", ".join(self._str(v) for v in val) + "]"
        if isinstance(val, SparkFunction):
            return str(val)
        return str(val)
    
    def eval(self, node: Any, env: Env) -> Any:
        # Dispatch based on node type
        node_type = type(node).__name__
        
        if node_type == 'Number':
            return node.value
        
        if node_type == 'String':
            return node.value
        
        if node_type == 'Boolean':
            return node.value
        
        if node_type == 'Name':
            return env.get(node.name)
        
        if node_type == 'List':
            elements = []
            for elem in node.elements:
                val = self.eval(elem, env)
                if missed(val):
                    return val
                elements.append(val)
            return elements

            obj = self.eval(node.obj, env)
            if missed(obj):
                return obj
            index = self.eval(node.index, env)
            if missed(index):
                return index
            
            if isinstance(obj, (list, str)):
                if not isinstance(index, int):
                    return Miss("index must be integer")

            return Miss("can only index list or string")

                return right
            if not isinstance(right, bool):
                return Miss("operands to && must be boolean")
            return right
        
        # Evaluate both operands for other ops
        left = self.eval(node.left, env)
        if missed(left):
            return left
        right = self.eval(node.right, env)
        if missed(right):
            return right
        
        if op == '+':
            if isinstance(left, (int, float)) and isinstance(right, (int, float)):
                return left + right
            if isinstance(left, str) and isinstance(right, str):
                return left + right
            if isinstance(left, list) and isinstance(right, list):
                return left + right
            return Miss(f"cannot add {type(left).__name__} and {type(right).__name__}")
        
        if op == '-':
            if isinstance(left, (int, float)) and isinstance(right, (int, float)):
                return left - right
            return Miss("- requires numbers")
        
        if op == '*':
            if isinstance(left, (int, float)) and isinstance(right, (int, float)):
                return left * right
            return Miss("* requires numbers")
        
        if op == '/':
            if not isinstance(left, (int, float)) or not isinstance(right, (int, float)):
                return Miss("/ requires numbers")
            if right == 0:
                return Miss("division by zero")
            return left / right
        
        if op == '%':
            if not isinstance(left, (int, float)) or not isinstance(right, (int, float)):
                return Miss("% requires numbers")
            if right == 0:
                return Miss("modulo by zero")
            return left % right
        
        if op == '==':
            return left == right
        
        if op == '!=':
            return left != right
        
        if op == '<':
            if isinstance(left, (int, float)) and isinstance(right, (int, float)):
                return left < right
            return Miss("< requires numbers")
        
        if op == '<=':
            if isinstance(left, (int, float)) and isinstance(right, (int, float)):
                return left <= right
            return Miss("<= requires numbers")
        
        if op == '>':
            if isinstance(left, (int, float)) and isinstance(right, (int, float)):
                return left > right
            return Miss("> requires numbers")
        
        if op == '>=':
            if isinstance(left, (int, float)) and isinstance(right, (int, float)):
                return left >= right
            return Miss(">= requires numbers")
        
        return Miss(f"unknown operator {op}")
    
    def _eval_unaryop(self, node: UnaryOp, env: Env) -> Any:
        operand = self.eval(node.operand, env)
        if missed(operand):
            return operand
        
        if node.op == '!':
            if not isinstance(operand, bool):
                return Miss("! requires boolean")
            return not operand
        
        if node.op == '-':
            if not isinstance(operand, (int, float)):
                return Miss("- requires number")
            return -operand
        
        return Miss(f"unknown unary operator {node.op}")
'''

with open('evaluator.py', 'w') as f:
    f.write(evaluator_code)

run_code = '''#!/usr/bin/env python3
"""Spark language interpreter."""
import sys
import os

# Make imports work from any directory
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from parser import parse, ParseError
from evaluator import Evaluator, Miss

def main():
    """Run Spark interpreter. Exit codes: 0 success, 1 runtime error, 2 parse error."""
    if len(sys.argv) != 2:
        print("usage: run.py <file.spark>", file=sys.stderr)
        sys.exit(2)
    
    filepath = sys.argv[1]
    
    try:
        with open(filepath, 'r') as f:
            src = f.read()
    except FileNotFoundError:
        print(f"error: file not found: {filepath}", file=sys.stderr)
        sys.exit(2)
    except Exception as e:
        print(f"error: {e}", file=sys.stderr)
        sys.exit(2)
    
    # Parse
    try:
        ast = parse(src)
    except ParseError as e:
        print(str(e), file=sys.stderr)
        sys.exit(2)
    except Exception as e:
        print(f"error: {e}", file=sys.stderr)
        sys.exit(2)
    
    # Evaluate
    try:
        evaluator = Evaluator()
        from evaluator import Env
        env = Env()
        result = evaluator.eval(ast, env)
        
        # Check for uncaught errors
        if isinstance(result, Miss):
            print(f"Miss({result.reason})", file=sys.stdout)
            sys.exit(1)
        
        sys.exit(0)
    except RecursionError:
        print("error: stack overflow", file=sys.stderr)
        sys.exit(1)
    except Exception as e:
        print(f"error: {e}", file=sys.stderr)
        sys.exit(1)

if __name__ == '__main__':
    main()
'''
