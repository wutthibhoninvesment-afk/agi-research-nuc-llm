# Calc Language Specification

## Overview
Calc is a small imperative scripting language with arithmetic, variables, functions, control flow, and lists.

## Design Decisions
- **Errors as values**: Runtime errors propagate as `Miss` values instead of exceptions
- **Newline sensitivity**: Newlines separate statements; suppressed inside brackets
- **First-class functions**: Functions are values; closures capture their definition environment
- **Dynamic typing**: Values include numbers, bools, lists, functions, and error values

## Grammar (Precedence: low to high)
```
program    → stmt*
stmt       → expr NEWLINE | block
block      → LBRACE stmt* RBRACE
expr       → assign
assign     → NAME EQUALS expr | logic_or
logic_or   → logic_and (OR logic_and)*
logic_and  → equality (AND equality)*
equality   → comparison ((EQ | NE) comparison)*
comparison → additive ((LT | LE | GT | GE) additive)*
additive   → multiply ((PLUS | MINUS) multiply)*
multiply   → unary ((STAR | SLASH | PERCENT) unary)*
unary      → (NOT | MINUS) unary | call
call       → primary (LPAREN args RPAREN)*
primary    → NUMBER | TRUE | FALSE | STRING | NAME | LPAREN expr RPAREN | list | fn | if
list       → LBRACKET (expr (COMMA expr)*)? RBRACKET
fn         → FN LPAREN params RPAREN block
if         → IF expr block ELSE block
args       → (expr (COMMA expr)*)?
params     → (NAME (COMMA NAME)*)?
```

## Runtime Errors (as Miss values)
- Unbound name
- Wrong arity in function call
- Division by zero
- Non-number in arithmetic
- Non-bool in condition
- Invalid index/field access
- Type mismatch

## Exit Codes
- 0: Success
- 1: Program-level error (runtime miss)
- 2: Lexer or parser error

- `len(x)`: list/string length
- `push(list, val)`: append to list (mutates)

- `print(val, ...)`: output to stdout

class Node:
    pass

def _simple(name, fields):
    class C(Node):
        def __init__(self, *args, **kwargs):
            self.line = kwargs.pop('line', 0)
            for f, v in zip(fields, args):
                setattr(self, f, v)
            for k, v in kwargs.items():
                setattr(self, k, v)
        def __repr__(self):
            return f"{name}({', '.join(getattr(self, f, None) for f in fields)})"
    C.__name__ = name
    return C

class Parser:
    def __init__(self, tokens):
        self.tokens = tokens
        self.pos = 0
        self.depth = 0
        self.max_depth = 1000
    
    def error(self, msg):
        tok = self.current()
        raise ParseError(msg, tok.line, tok.col)
    
    def current(self):
        return self.tokens[min(self.pos, len(self.tokens) - 1)]
    
    def peek(self, offset=1):
        return self.tokens[min(self.pos + offset, len(self.tokens) - 1)]
    
    def advance(self):
        self.pos += 1
    
    def skip_newlines(self):
        while self.current().type == TokenType.NEWLINE:
            self.advance()
    
    def expect(self, tt):
        if self.current().type != tt:
            self.error(f'expected {tt}, got {self.current().type}')
        tok = self.current()
        self.advance()
        return tok
    
    def parse(self):
        stmts = []
        self.skip_newlines()
        while self.current().type != TokenType.EOF:
            stmts.append(self.stmt())
            self.skip_newlines()
        return Program(stmts, line=self.tokens[0].line)
    
    def stmt(self):
        if self.current().type == TokenType.LBRACE:
            return self.block()
        expr = self.expr()
        if self.current().type == TokenType.NEWLINE:
            self.advance()
        return expr
    
    def block(self):
        self.expect(TokenType.LBRACE)
        stmts = []
        self.skip_newlines()
        while self.current().type != TokenType.RBRACE:
            stmts.append(self.stmt())
            self.skip_newlines()
        self.expect(TokenType.RBRACE)
        return Block(stmts, line=self.tokens[self.pos - 1].line)
    
    def expr(self):
        return self.assign()
    
    def assign(self):
        expr = self.logic_or()
        if self.current().type == TokenType.EQUALS and isinstance(expr, Name):
            self.advance()
            value = self.assign()
            return Assign(expr.name, value, line=expr.line)
        return expr
    
    def logic_or(self):
        left = self.logic_and()
        while self.current().type == TokenType.OR:
            op_tok = self.current()
            self.advance()
            right = self.logic_and()
            left = BinOp(TokenType.OR, left, right, line=op_tok.line)
        return left
    
    def logic_and(self):
        left = self.equality()
        while self.current().type == TokenType.AND:
            op_tok = self.current()
            self.advance()
            right = self.equality()
            left = BinOp(TokenType.AND, left, right, line=op_tok.line)
        return left
    
    def equality(self):
        left = self.comparison()
        while self.current().type in (TokenType.EQ, TokenType.NE):
            op = self.current().type
            op_tok = self.current()
            self.advance()
            right = self.comparison()
            left = BinOp(op, left, right, line=op_tok.line)
        return left
    
    def comparison(self):
        left = self.additive()
        while self.current().type in (TokenType.LT, TokenType.LE, TokenType.GT, TokenType.GE):
            op = self.current().type
            op_tok = self.current()
            self.advance()
            right = self.additive()
            left = BinOp(op, left, right, line=op_tok.line)
        return left
    
    def additive(self):
        left = self.multiply()
        while self.current().type in (TokenType.PLUS, TokenType.MINUS):
            op = self.current().type
            op_tok = self.current()
            self.advance()
            right = self.multiply()
            left = BinOp(op, left, right, line=op_tok.line)
        return left
    
    def multiply(self):
        left = self.unary()
        while self.current().type in (TokenType.STAR, TokenType.SLASH, TokenType.PERCENT):
            op = self.current().type
            op_tok = self.current()
            self.advance()
            right = self.unary()
            left = BinOp(op, left, right, line=op_tok.line)
        return left
    
    def unary(self):
        if self.current().type in (TokenType.NOT, TokenType.MINUS):
            op = self.current().type
            op_tok = self.current()
            self.advance()
            operand = self.unary()
            return UnaryOp(op, operand, line=op_tok.line)
        return self.call()
    
    def call(self):
        expr = self.primary()
        while self.current().type == TokenType.LPAREN:
            self.advance()
            args = []
            if self.current().type != TokenType.RPAREN:
                args.append(self.expr())
                while self.current().type == TokenType.COMMA:
                    self.advance()
                    args.append(self.expr())
            self.expect(TokenType.RPAREN)
            expr = Call(expr, args, line=expr.line)
        return expr
    
    def primary(self):
        tok = self.current()
        
        if tok.type == TokenType.NUMBER:
            self.advance()
            return Number(tok.value, line=tok.line)
        
        if tok.type == TokenType.STRING:
            self.advance()
            return String(tok.value, line=tok.line)
        
        if tok.type == TokenType.TRUE:
            self.advance()
            return Bool(True, line=tok.line)
        
        if tok.type == TokenType.FALSE:
            self.advance()
            return Bool(False, line=tok.line)
        
        if tok.type == TokenType.NAME:
            self.advance()
            return Name(tok.value, line=tok.line)
        
        if tok.type == TokenType.LBRACKET:
            return self.list_literal()
        
        if tok.type == TokenType.FN:
            return self.fn_literal()
        
        if tok.type == TokenType.IF:
            return self.if_expr()
        
        if tok.type == TokenType.LPAREN:
            self.advance()
            expr = self.expr()
            self.expect(TokenType.RPAREN)
            return expr
        
        self.error(f'unexpected {tok.type}')
    
    def list_literal(self):
        start_tok = self.current()
        self.expect(TokenType.LBRACKET)
        elements = []
        if self.current().type != TokenType.RBRACKET:
            elements.append(self.expr())
            while self.current().type == TokenType.COMMA:
                self.advance()
                if self.current().type == TokenType.RBRACKET:
                    break
                elements.append(self.expr())
        self.expect(TokenType.RBRACKET)

    def fn_literal(self):
        start_tok = self.current()
        self.expect(TokenType.FN)
        self.expect(TokenType.LPAREN)

            params.append(self.current().value)

                self.advance()
                params.append(self.expect(TokenType.NAME).value)
        self.expect(TokenType.RPAREN)
        body = self.block()
        return Fn(params, body, line=start_tok.line)
    
    def if_expr(self):
        start_tok = self.current()
        self.expect(TokenType.IF)
        cond = self.expr()
        then_block = self.block()
        self.expect(TokenType.ELSE)
        else_block = self.block()

class Miss:
    def __init__(self, reason):
        self.reason = reason
    
    def __repr__(self):
        return f"Miss({self.reason!r})"
    
    def __eq__(self, other):
        return isinstance(other, Miss) and self.reason == other.reason
    
    def __hash__(self):
        return hash(self.reason)

def is_miss(v):
    return isinstance(v, Miss)

class Closure:
    def __init__(self, params, body, env):
        self.params = params
        self.body = body
        self.env = env
    
    def __repr__(self):
        return f"<fn({', '.join(self.params)})>"

class Env:
    def __init__(self, parent=None):
        self.parent = parent
        self.bindings = {}
    
    def define(self, name, value):
        self.bindings[name] = value
    
    def get(self, name):
        if name in self.bindings:
            return self.bindings[name]
        if self.parent:
            return self.parent.get(name)
        return Miss(f'unbound name "{name}"')
    
    def set(self, name, value):
        if name in self.bindings:
            self.bindings[name] = value
            return value
        if self.parent:
            return self.parent.set(name, value)
        self.bindings[name] = value
        return value

class Evaluator:
    def __init__(self):

            env = self.global_env
        
        if node is None:
            return None
        
        node_type = node.__class__.__name__
        if node_type in self.dispatch:
            return self.dispatch[node_type](node, env)
        
        return Miss(f'unknown node type {node_type}')

        result = None
        for stmt in node.stmts:
            result = self.eval(stmt, env)
            if is_miss(result):
                return result
        return result if result is not None else 0
    
    def eval_block(self, node, env):
        result = None
        for stmt in node.stmts:
            result = self.eval(stmt, env)
            if is_miss(result):
                return result
        return result if result is not None else 0
    
    def eval_number(self, node, env):
        return node.value
    
    def eval_string(self, node, env):
        return node.value

            if is_miss(val):

        if node.op == TokenType.LE:
            if isinstance(left, (int, float)) and isinstance(right, (int, float)):
                return left <= right
            return Miss('type mismatch in <=')
        
        if node.op == TokenType.GT:
            if isinstance(left, (int, float)) and isinstance(right, (int, float)):
                return left > right
            return Miss('type mismatch in >')
        
        if node.op == TokenType.GE:
            if isinstance(left, (int, float)) and isinstance(right, (int, float)):
                return left >= right
            return Miss('type mismatch in >=')
        
        if node.op == TokenType.EQ:
            return left == right
        
        if node.op == TokenType.NE:
            return left != right
        
        if node.op == TokenType.AND:
            if not isinstance(left, bool) or not isinstance(right, bool):
                return Miss('type mismatch in and')
            return left and right
        
        if node.op == TokenType.OR:
            if not isinstance(left, bool) or not isinstance(right, bool):
                return Miss('type mismatch in or')
            return left or right
        
        return Miss(f'unknown operator {node.op}')
    
    def eval_unaryop(self, node, env):
        operand = self.eval(node.operand, env)
        if is_miss(operand):
            return operand
        
        if node.op == TokenType.NOT:
            if not isinstance(operand, bool):
                return Miss('type mismatch in not')
            return not operand
        
        if node.op == TokenType.MINUS:
            if isinstance(operand, (int, float)):
                return -operand
            return Miss('type mismatch in unary -')
        
        return Miss(f'unknown unary operator {node.op}')
    
    def eval_assign(self, node, env):
        value = self.eval(node.value, env)
        if is_miss(value):
            return value
        env.set(node.name, value)
        return value
    
    def eval_call(self, node, env):
        func = self.eval(node.func, env)
        if is_miss(func):
            return func
        
        args = []
        for arg_node in node.args:
            arg = self.eval(arg_node, env)
            if is_miss(arg):
                return arg
            args.append(arg)
        
        if callable(func) and not isinstance(func, Closure):
            return func(*args)
        
        if isinstance(func, Closure):
            if len(args) != len(func.params):
                return Miss(f'wrong arity: expected {len(func.params)}, got {len(args)}')
            
            call_env = Env(func.env)
            for param, arg in zip(func.params, args):
                call_env.define(param, arg)
            
            return self.eval(func.body, call_env)
        
        return Miss(f'not a function')
    
    def eval_if(self, node, env):
        cond = self.eval(node.cond, env)
        if is_miss(cond):
            return cond
        
        if not isinstance(cond, bool):
            return Miss('condition must be boolean')
        
        if cond:
            return self.eval(node.then_block, env)
        else:
            return self.eval(node.else_block, env)
    
    def eval_fn(self, node, env):
        return Closure(node.params, node.body, env)
    
    def builtin_len(self, x):
        if isinstance(x, (list, str)):
            return len(x)
        return Miss('len() requires list or string')
    
    def builtin_print(self, *args):
        for arg in args:
            if isinstance(arg, bool):
                print(str(arg).lower(), end=' ')
            else:
                print(arg, end=' ')
        print()
        return 0
